# Dolice Labs unified robots.txt — minimal Disallow policy
#
# Why minimal: HTML <meta name="robots" content="noindex"> can only be honored
# if Google can crawl the page. Combining robots.txt Disallow with noindex is
# self-defeating — blocked URLs that are linked externally end up indexed
# (URL-only) because Google never reads the noindex tag inside.
#
# Pages we don't want indexed (/tag/, /tags, ?page=, ?category=, ?thanks=)
# rely on HTML <meta name="robots" content="noindex"> + rel="canonical"
# instead of robots.txt Disallow. This is the official Google guidance.
# (https://developers.google.com/search/docs/crawling-indexing/block-indexing)
#
# Only Disallow:
#   (a) server-only endpoints that never need to be crawled
#   (b) URLs carrying sensitive query tokens (Stripe session_id)
#   (c) common 404 probe paths (login/wp-admin/.env etc.)

User-agent: *
Allow: /

# Server-only endpoints — never linked, never need crawling
Disallow: /api/
Disallow: /auth/

# Stripe checkout return URL — contains sensitive session token
Disallow: /*?session_id=

# Common 404 probe paths — no real content here, saves crawl budget
Disallow: /login
Disallow: /register
Disallow: /signup
Disallow: /signin
Disallow: /logout
Disallow: /profile
Disallow: /dashboard
Disallow: /settings
Disallow: /account
